1. Add thermostat and meter status to configuration.yaml

  mqtt:
  binary_sensor:
    - name: "PV box fan cooling"
      state_topic: "solar_lan/thermostat/status"
      payload_on: true
      payload_off: false
      value_template: "{{ value_json.cooling }}"
      unique_id: "thermostat_pv_cooling"
    - name: "PV box fan heating"
      state_topic: "solar_lan/thermostat/status"
      payload_on: true
      payload_off: false
      value_template: "{{ value_json.heating }}"
      unique_id: "thermostat_pv_heating"

  sensor:
    - name: "PV box internal temperature"
      state_topic: "solar_lan/thermostat/status"
      unit_of_measurement: "℃"
      value_template: '{{ value_json.temp }}'
      unique_id: "thermostat_pv_temperature"
    - name: "PV box internal humidity"
      state_topic: "solar_lan/thermostat/status"
      unit_of_measurement: "%"
      value_template: '{{ value_json.hum }}'
      unique_id: "thermostat_pv_humidity"
    - name: "PV box meter current"
      state_topic: "solar_lan/powermeter_bl/status"
      unit_of_measurement: "A"
      value_template: '{{ value_json.current }}'
      unique_id: "powermeter_pv_current"
    - name: "PV box meter voltage"
      state_topic: "solar_lan/powermeter_bl/status"
      unit_of_measurement: "V"
      value_template: '{{ value_json.voltage }}'
      unique_id: "powermeter_pv_voltage"
    - name: "PV box meter power"
      state_topic: "solar_lan/powermeter_bl/status"
